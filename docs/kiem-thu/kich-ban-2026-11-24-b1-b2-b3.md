-- (f) số sản phẩm đang bán của shop có voucher SHOP còn hiệu lực (khớp D5)

   SELECT DISTINCT vendor_id FROM vouchers WHERE type = 'SHOP' AND is_active AND vendor_id IS NOT NULL
